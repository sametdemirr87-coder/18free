    let captchaAlarmContext=null;
    let captchaAlarmPlaying=false;
    function primeCaptchaAlarm(){
        try{
            const AudioContextClass=window.AudioContext||window.webkitAudioContext;
            if(!AudioContextClass)return;
            if(!captchaAlarmContext)captchaAlarmContext=new AudioContextClass();
            if(captchaAlarmContext.state==='suspended')captchaAlarmContext.resume().catch(()=>{});
        }catch{}
    }
    function playCaptchaAlarm(){
        if(captchaAlarmPlaying)return;
        primeCaptchaAlarm();
        const context=captchaAlarmContext;
        if(!context)return;
        const start=()=>{
            if(captchaAlarmPlaying)return;
            captchaAlarmPlaying=true;
            const base=context.currentTime+.04;
            for(let index=0;index<3;index++){
                const oscillator=context.createOscillator();
                const gain=context.createGain();
                const at=base+(index*.48);
                oscillator.type='square';
                oscillator.frequency.setValueAtTime(index===1?1120:920,at);
                gain.gain.setValueAtTime(.0001,at);
                gain.gain.exponentialRampToValueAtTime(.38,at+.025);
                gain.gain.exponentialRampToValueAtTime(.0001,at+.3);
                oscillator.connect(gain);gain.connect(context.destination);
                oscillator.start(at);oscillator.stop(at+.32);
            }
            setTimeout(()=>{captchaAlarmPlaying=false;},1750);
        };
        if(context.state==='running')start();
        else context.resume().then(start).catch(()=>{});
    }

    function telegramServerRequest(method,path,payload={}){
        const client=nexusClient();
        const base=String(client.serverUrl||'').replace(/\/$/,'');
        if(!base||!client.sessionToken)return Promise.resolve({success:false,error:'Nexus lisans oturumu hazir degil'});

        return new Promise(resolve=>{
            GM_xmlhttpRequest({method,url:base+'/cryptocrops'+path,headers:{'Content-Type':'application/json'},data:JSON.stringify({...auth,...payload}),timeout:25000,
                onload:response=>{try{const parsed=JSON.parse(response.responseText||'{}');resolve(response.status>=200&&response.status<300?parsed:{success:false,error:parsed.detail||parsed.error||('HTTP '+response.status)});}catch{resolve({success:false,error:'Sunucu yaniti okunamadi'});}},

    function enableDrag(element, key){
        const header = element.querySelector('.nx5-head');
        let drag = null;
        header.addEventListener('pointerdown', event => {
            if(event.button !== 0 || event.target.closest('button')) return;
            const rect = element.getBoundingClientRect();

            header.setPointerCapture?.(event.pointerId); bringFront(element); event.preventDefault();
        });
        header.addEventListener('pointermove', event => {
            if(!drag || event.pointerId !== drag.id) return;
            element.style.left = `${Math.max(0, Math.min(window.innerWidth - element.offsetWidth, drag.left + event.clientX - drag.x))}px`;

            element.style.right = 'auto';
        });
        header.addEventListener('pointerup', event => { if(drag && event.pointerId === drag.id){ drag = null; savePosition(element,key); } });
        header.addEventListener('pointercancel', () => { drag = null; });
        element.addEventListener('pointerdown', () => bringFront(element));

    async function api(url,options={}){
        const response=await fetch(url,{credentials:'same-origin',cache:'no-store',...options}); let data={};
        try{data=await response.json();}catch{}
        if(!response.ok){
            const error=Error(`${url} HTTP ${response.status}: ${data.message||data.error||'istek reddedildi'}`);error.code=data.error||`HTTP_${response.status}`;
            if(error.code==='FARM_CHECKPOINT_REQUIRED')requestFarmCheckpoint();
            throw error;
        }
        return data;
    }
    async function load(sayLog=false){
        try{
            const data=await api('/api/bootstrap'); state.farm=data.farm||data; state.homestead=data.homestead||null; state.offset=Number(state.farm.server_now||Date.now())-Date.now();
            if(state.checkpointPending&&state.farm?.farm_checkpoint&&!state.farm.farm_checkpoint.required){state.checkpointPending=false;state.checkpointNotified=false;log(t('checkpointPassed'),'ok');}
            const unknownOwned=(state.farm.seeds||[]).some(seed=>!state.catalog.has(seed.seed_id||seed.id));
            if(!state.catalog.size||unknownOwned){const catalog=await api('/api/catalog?version=3');state.catalog=new Map((catalog.seeds||[]).map(seed=>[seed.seed_id,seed]));}
            state.lastLoadAt=Date.now();
            if(sayLog)log(t('dataRefreshed'),'info'); refreshOpenWindows(); return true;
        }catch(error){log(t('dataError',{error:error.message}),'bad');return false;}
    }

    function orderSeedPlan(){
        if(!state.autoOrders)return {};
        const order=(state.farm?.orders||[]).find(entry=>!entry.ready&&(entry.requirements||[]).every(req=>Number(ownedSeed(req.seed_id)?.quantity||0)>0));if(!order)return {};
        const plan={};for(const req of order.requirements||[]){const growing=plantedCopies(req.seed_id);const missing=Math.max(0,Number(req.quantity||0)-inventoryQty(req.seed_id)-growing);if(missing>0)plan[req.seed_id]=missing;}return plan;
    }
    function chooseSeedForPlot(plot,orderPlan){
        for(const [seed,count] of Object.entries(orderPlan)){if(count>0&&freeCopies(seed)>0){orderPlan[seed]-=1;return seed;}}
        if(state.autoPlots.has(plot))return [...ownedSeeds()].filter(seed=>freeCopies(seed.seed_id||seed.id)>0).sort((left,right)=>freeCopies(right.seed_id||right.id)-freeCopies(left.seed_id||left.id)||Number(seedInfo(left.seed_id||left.id)?.grow_seconds||Infinity)-Number(seedInfo(right.seed_id||right.id)?.grow_seconds||Infinity))[0]?.seed_id||[...ownedSeeds()].find(seed=>freeCopies(seed.id)>0)?.id||null;
        const selected=state.plotSeeds[plot];return selected&&freeCopies(selected)>0?selected:null;
    }

    async function runCycle(){

        const readyCrop=crops().find(isReady);if(readyCrop){await harvest(readyCrop);if(state.running)await plantEmptyPlots();return;}
        await plantEmptyPlots();if(state.autoSell)await autoSell();
    }

    function finishFaucetClaim(){

        state.faucetClaimPending=false;state.faucetVerifiedAt=0;state.faucetConfirming=false;

        state.faucetPopup=null;
        state.faucetReadyKey=String(state.farm?.faucet?.next_claim_at||'claimed');

    }
    async function confirmFaucetClaim(){
        if(state.faucetConfirming||state.destroyed)return;
        state.faucetConfirming=true;
        const previousNextAt=Number(state.farm?.faucet?.next_claim_at||0);
        try{

                await wait(attempt?2000:800);
                await load();
                const nextAt=Number(state.farm?.faucet?.next_claim_at||0);
                if(!faucetIsReady()||(nextAt>previousNextAt&&nextAt>Date.now()+state.offset)){
                    finishFaucetClaim();return;
                }
            }
            state.faucetClaimPending=false;state.faucetVerifiedAt=0;
            if(state.faucetArmed&&faucetIsReady()&&!state.destroyed)retryFaucetPopup();
        }finally{state.faucetConfirming=false;}
    }
    function onMessage(event){
        if(event.origin!==location.origin)return;
        if(event.data?.source==='nexus-farm-checkpoint'){
            if(event.data.type==='opened'){playCaptchaAlarm();return;}
            if(event.data.type==='telegram-captcha'){void sendCaptchaToPrivateTelegram(event.data.kind||'farm',String(event.data.site_key||''),event.source);return;}
            if(event.data.type==='retry'){retryFarmCheckpoint();return;}
            if(event.data.type==='challenge'){log(t('visualChallenge'),'info');return;}
            if(event.data.type==='passed'){
                clearTimeout(state.checkpointRetryTimer);state.checkpointRetryTimer=null;state.checkpointRetryCount=0;
                try{if(state.checkpointPopup&&!state.checkpointPopup.closed)state.checkpointPopup.close();}catch{}
                state.checkpointPopup=null;state.checkpointProbeAt=Date.now();
                void confirmFarmCheckpoint();
            }
            return;
        }
        if(event.data?.source!=='nexus-faucet')return;
        if(event.data.type==='opened'){playCaptchaAlarm();log(t('captchaWaiting'),'info');return;}
        if(event.data.type==='telegram-captcha'){void sendCaptchaToPrivateTelegram(event.data.kind||'faucet',String(event.data.site_key||''),event.source);return;}
        if(event.data.type==='retry'){retryFaucetPopup();return;}
        if(event.data.type==='challenge'){log(t('visualChallenge'),'info');return;}
        if(event.data.type==='verified'){
            clearTimeout(state.faucetRetryTimer);state.faucetRetryTimer=null;
            state.faucetClaimPending=true;state.faucetVerifiedAt=Date.now();
            log(t('verified'),'ok');void confirmFaucetClaim();return;
        }
        if(event.data.type==='claimed'){finishFaucetClaim();return;}
        log(t('captchaWaiting'),'info');
    }
    function tick(){
        document.querySelectorAll('[data-at]').forEach(element=>element.textContent=formatLeft(element.dataset.at));
        document.querySelectorAll('[data-faucet-at]').forEach(element=>element.textContent=faucetIsReady()?t('ready'):formatLeft(element.dataset.faucetAt));
        if(state.checkpointPopup&&!state.checkpointPopup.closed&&popupWasRedirected(state.checkpointPopup))retryFarmCheckpoint();
        if(state.faucetPopup&&!state.faucetPopup.closed&&!state.faucetClaimPending&&popupWasRedirected(state.faucetPopup))retryFaucetPopup();
        if(checkpointVisible()&&!state.checkpointPending)requestFarmCheckpoint();
        if(state.checkpointPending&&!state.checkpointConfirming&&!state.checkpointRetryTimer&&(!state.checkpointPopup||state.checkpointPopup.closed)&&Date.now()>=state.checkpointProbeAt){state.checkpointProbeAt=Date.now()+15000;void confirmFarmCheckpoint();}
        if(state.faucetArmed&&faucetIsReady()&&!state.faucetClaimPending){
            const key=String(state.farm?.faucet?.next_claim_at||'ready');
            if(state.faucetReadyKey!==key){state.faucetReadyKey=key;openFaucetPopup(true);}
        }else if(!faucetIsReady())state.faucetReadyKey='';
    }

    }
