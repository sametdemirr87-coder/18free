  function getBearerToken() {

  async function callSupabaseRpc(fnName, payload) {

      throw new Error('No Bearer Token');

      headers: {
        'accept': '*/*',
        'apikey': API_KEY,
        'authorization': `Bearer ${token}`,
        'content-profile': 'public',
        'content-type': 'application/json',
        'x-client-info': 'supabase-js-web/2.90.1'
      },
      body: JSON.stringify(payload)
    });

    if (!res.ok) {
      const errTxt = await res.text();
      throw new Error(`HTTP ${res.status}: ${errTxt}`);
    }
    return await res.json();
  }

  async function executeGameFlow(game) {
    if (isPlaying) {
      log('Hali hazırda çalışan bir oturum var.', 'warn');
      return;
    }
    if (Math.max(cooldowns[game.id] || 0, retryAfter[game.id] || 0) > Date.now()) { log(game.name + ' · Bekleme süresi devam ediyor.', 'warn'); return; }

    updateStatus('Starting', game.name);

    try {
      // 1. Supabase: start_game_session_secure
      log(`Oyun başlatılıyor · ${game.name}`, 'play');
      const startRes = await callSupabaseRpc('start_game_session_secure', {
        p_game_name: game.id
      });

      if (!startRes || !startRes.success) {
        throw new Error(startRes?.message || 'Session başlatılamadı');
      }

      const { session_id, nonce, target, min_duration } = startRes;
      log(`Oturum açıldı · ${game.name} · Hedef skor: ${target}`, 'ok');

      // Sunucudan öğrenilen süre veya kullanıcı aralığı; sunucu alt sınırı korunur.
      const waitSeconds = chooseDuration(game, startRes);

      updateStatus('Playing', `${game.name} (${waitSeconds}s)`);
      log(`Oynanıyor... ${waitSeconds} saniye bekleniyor.`, 'play');
      await runCountdown(waitSeconds, game.name);

      // 3. Supabase: claim_game_session_secure
      const safeScore = (target || game.defaultTarget) + randInt(5, 15);
      updateStatus('Claiming', game.name);
      log(`Oyun tamamlanıyor · ${game.name} · Skor: ${safeScore}`, 'sys');

      const claimRes = await callSupabaseRpc('claim_game_session_secure', {
        p_session_id: session_id,
        p_nonce: nonce,
        p_score: safeScore
      });

      if (claimRes && claimRes.success) {
        log(`Oyun bitti · ${game.name} · +${claimRes.power_earned} Power · Seviye ${claimRes.new_level}`, 'ok');
        await recordGameResult(game, claimRes);
      } else {
        retryAfter[game.id] = Date.now() + 60000;
        log(game.name + ' · ' + (claimRes?.error === 'Target not reached' ? 'Hedef tamamlanmadı; ödül alınamadı.' : 'Oturum onaylanmadı; ödül alınamadı.'), 'warn');
      }

    } catch (err) {
      retryAfter[game.id] = Date.now() + 60000;
      log(`Hata: ${err.message}`, 'err');
    } finally {
      isPlaying = false; activeGameId = '';
      await refreshCooldowns();
      clearCountdown();
      if (!isRunning) updateStatus('Idle', 'Hazır');
    }
  }

  // --- Zamanlayıcı ve Durum Kontrolleri ---
  function runCountdown(sec, label) {
    clearCountdown();
    countdownTotal = Math.max(1, sec);
    const deadline = Date.now() + sec * 1000;
    return new Promise(resolve => {
      countdownResolve = resolve;
      const tick = () => {
        const left = Math.max(0, (deadline - Date.now()) / 1000);
        renderTimer(left, label);
        if (left <= 0) clearCountdown();
      };
      activeTimer = setInterval(tick, 100);
      tick();
    });
  }
  function clearCountdown() {
    clearInterval(activeTimer);
    activeTimer = null;
    const resolve = countdownResolve;
    countdownResolve = null;
    renderTimer(0, '');
    if (resolve) resolve();
  }
  function renderTimer(left, label) {
    if (!ui.timer) return;

    const text = String(Math.floor(seconds / 60)).padStart(2, '0') + ':' + String(seconds % 60).padStart(2, '0');

    ui.clock.style.setProperty('--p', (left > 0 ? left / Math.max(1, countdownTotal) * 100 : 0) + '%');
    if (label) ui.substatus.textContent = label;
  }

  function missionTarget(m) {
    let target = m.requirement_target;
    if (target?.startsWith('rotating_')) {
      const now = new Date();
      const day = Math.floor((Date.UTC(now.getUTCFullYear(),now.getUTCMonth(),now.getUTCDate()) - Date.UTC(now.getUTCFullYear(),0,0)) / 86400000);
      target = ['clicker','memory','math','puzzle','ufo','catch','blockbreaker','match3'][(day + (parseInt(target.split('_')[1]) || 1)) % 8];
    }
    return target;
  }
  function selectGame() {
    const available = GAMES.filter(game => Math.max(cooldowns[game.id] || 0, retryAfter[game.id] || 0) <= Date.now());
    if (!available.length) return null;
    const missionIds = new Set();
    if (prefs.smart && missionsFresh) {
      for (const row of missionRows) {
        if (row.available && !row.is_completed && !row.is_claimed &&
            (!row.reset_at || Date.parse(row.reset_at) > Date.now()) &&
            ['win_game','reach_level'].includes(row.mission.requirement_type) &&
            Number(row.mission.requirement_value) > Number(row.estimated ?? row.progress ?? 0)) {
          missionIds.add(missionTarget(row.mission));
        }
      }
    }
    const priority = available.filter(g => missionIds.has(g.id));
    const other = available.filter(g => !missionIds.has(g.id));
    // 60/40 applies to eligible groups, not to each individual mission row.
    let pool = priority.length && other.length ? (Math.random() < 0.60 ? priority : other) : available;
    const different = pool.filter(g => g.id !== lastSelectedGame);
    if (different.length) pool = different;
    else if (available.length > 1) pool = available.filter(g => g.id !== lastSelectedGame);
    // Least-recently selected within each group prevents starvation of other games.
    const oldest = Math.min(...pool.map(g => selectionHistory.get(g.id) || 0));
    pool = pool.filter(g => (selectionHistory.get(g.id) || 0) === oldest);
    const game = pool[Math.floor(Math.random() * pool.length)];
    selectionHistory.set(game.id, ++selectionNumber);
    lastSelectedGame = game.id;
    return game;
  }
  async function recordGameResult(game, result) {
    if (!missionsFresh) return;
    let reached = false;
    for (const row of missionRows) {
      if (!row.available || row.is_completed || row.is_claimed) continue;
      const m = row.mission, target = missionTarget(m);
      if (m.requirement_type === 'total_wins' || (m.requirement_type === 'win_game' && target === game.id))
        row.estimated = Number(row.estimated ?? row.progress ?? 0) + 1;
      else if (m.requirement_type === 'reach_level' && target === game.id)
        row.estimated = Math.max(Number(row.progress) || 0, Number(result.new_level) || 0);
      if (row.estimated >= Number(m.requirement_value)) reached = true;
    }
    renderMissions();
    if (reached) { log('Görev hedefi kontrol ediliyor · Sunucu doğrulaması', 'sys'); await missionPulse(); }
  }
  function chooseDuration(game, response) {
    const time = Number(response.time_limit);
    const minimum = Math.max(1, Number(response.min_duration) || 1);
    const entry = prefs.games[game.id] || {min:20,max:40};
    if (prefs.autoTime) {
      if (!Number.isFinite(time) || time < minimum) throw new Error('Sunucu geçerli oyun süresi döndürmedi. Auto Süre kapatılarak manuel aralık kullanılabilir.');
      const level = Math.max(1, Number(response.level) || 1);
      const difficulty = Math.min(1, (level - 1) / 29);
      const lowerRatio = 0.45 + 0.40 * difficulty;
      entry.min = Math.min(time, Math.max(Math.ceil(minimum + 1), Math.ceil(time * lowerRatio)));
      entry.max = Math.floor(time); entry.learned = true; entry.serverMin = minimum; entry.level = level;
      prefs.games[game.id] = entry; saveState(); renderSettings();
    }
    const floor = Math.ceil(minimum);
    const ceiling = Number.isFinite(time) && time >= floor ? Math.floor(time) : Math.max(floor, entry.max);
    const low = Math.min(ceiling, Math.max(floor, entry.min));
    const high = Math.min(ceiling, Math.max(low, entry.max));
    return randInt(low, high);
  }

  async function refreshCooldowns() {
    if (cooldownLoading) return cooldownLoading;
    cooldownLoading = (async () => {
      try {
        const token = getBearerToken();
        if (!token) throw new Error('Giriş gerekli.');
        const part = token.split('.')[1].replace(/-/g,'+').replace(/_/g,'/');
        const userId = JSON.parse(atob(part.padEnd(Math.ceil(part.length / 4) * 4,'='))).sub;
        if (!userId) throw new Error('Aktif kullanıcı bulunamadı.');
        const rows = await readTable('game_cooldowns', new URLSearchParams({select:'game_name,cooldown_ends_at',user_id:'eq.'+userId}));
        if (!Array.isArray(rows)) throw new Error('Bekleme bilgisi alınamadı.');
        cooldowns = Object.fromEntries(rows.map(row => [row.game_name, Date.parse(row.cooldown_ends_at) || 0]));
        cooldownKnown = true;
      } catch (error) {
        cooldownKnown = false;
        log('Bekleme bilgisi alınamadı · ' + error.message, 'warn');
      } finally { renderCooldowns(); }
    })();
    try { await cooldownLoading; } finally { cooldownLoading = null; }
  }
  function renderCooldowns() {
    if (!ui.gameList) return;
    ui.gameList.querySelectorAll('[data-game]').forEach(card => {
      const id=card.dataset.game, left=Math.max(0, Math.ceil((Math.max(cooldowns[id] || 0, retryAfter[id] || 0) - Date.now())/1000));
      const playing = activeGameId === id;
      const remaining = playing ? activeSeconds : left;
      const text = playing ? 'Oynanıyor' : left ? 'Bekleme' : cooldownKnown ? 'Hazır' : 'Bekleme bilgisi yüklenmedi';
      const label = card.querySelector('.cooldown');
      if (label) {
        const status = label.querySelector('.card-state');
        const digits = label.querySelector('.card-digits');
        const clock = remaining > 0 ? String(Math.floor(remaining/60)).padStart(2,'0') + ':' + String(remaining%60).padStart(2,'0') : playing ? '00:00' : '--:--';
        if (status.textContent !== translate(text)) status.textContent = text;
        if (digits.textContent !== clock) digits.textContent = clock;
        label.dataset.mode = playing ? 'playing' : left ? 'cooldown' : 'ready';
        let fill = 0;
        if (playing) {
          fill = (countdownTotal - activeSeconds) / Math.max(1, countdownTotal);
        } else if (left > 0) {
          const end = cooldowns[id];
          let window = cooldownWindows.get(id);
          if (!window || window.end !== end) {
            window = {end, duration: Math.max(1, end - Date.now())};
            cooldownWindows.set(id, window);
          }
          fill = (end - Date.now()) / window.duration;
        } else {
          cooldownWindows.delete(id);
        }
        label.style.setProperty('--fill', Math.max(0, Math.min(1, fill)));
      }
      card.querySelector('button').disabled = left>0 || isPlaying || loopActive;
    });
  }

  const localizedNodes = new WeakMap();
  const localizedAttributes = new WeakMap();
  function translate(text) {
    if (prefs.language !== 'en') return text;
    let value = text;
    for (const [source,target] of ENGLISH) value = value.split(source).join(target);
    return value;
  }
  function applyLanguage() {
    if (!ui.root) return;
    ui.root.host.lang = prefs.language === 'en' ? 'en' : 'tr';
    const walker = document.createTreeWalker(ui.root, NodeFilter.SHOW_TEXT);
    while (walker.nextNode()) {
      const node=walker.currentNode;
      if (node.parentElement?.closest('style,script')) continue;
      const cached=localizedNodes.get(node);
      const source=cached && node.nodeValue === cached.rendered ? cached.source : node.nodeValue;
      const rendered=translate(source);
      localizedNodes.set(node,{source,rendered});
      if (node.nodeValue !== rendered) node.nodeValue=rendered;
    }
    ui.root.querySelectorAll('[aria-label],[title]').forEach(el => {
      const cached=localizedAttributes.get(el) || {};
      for (const name of ['aria-label','title']) {
        if (!el.hasAttribute(name)) continue;
        const value=el.getAttribute(name), entry=cached[name];
        const source=entry && value===entry.rendered ? entry.source : value;
        const rendered=translate(source);
        cached[name]={source,rendered};
        if(value!==rendered) el.setAttribute(name,rendered);
      }
      localizedAttributes.set(el,cached);
    });
  }
  function startLocalization() {
    let queued=false;
    new MutationObserver(() => {
      if(queued) return;
      queued=true;
      queueMicrotask(()=>{queued=false;applyLanguage();});
    }).observe(ui.root,{subtree:true,childList:true,characterData:true});
    applyLanguage();
  }

  function escapeHtml(value) {
    return String(value ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
  }
  async function readTable(table, query) {
    const token = getBearerToken();
    if (!token) throw new Error('Görevleri görmek için siteye giriş yapın.');
    const response = await fetch(SUPABASE_BASE.replace(/\/rpc$/, '/') + table + '?' + query, {
      headers: { apikey: API_KEY, authorization: 'Bearer ' + token, accept: 'application/json' }
    });
    if (!response.ok) throw new Error('Görevler yüklenemedi (HTTP ' + response.status + ').');
    return response.json();
  }
  async function refreshMissions() {
    if (missionLoading) return;
    missionLoading = true;
    missionsFresh = false;
    renderMissions();
    ui.missionMessage.textContent = 'Görevler güncelleniyor…';
    ui.refresh.disabled = true;
    try {
      const token = getBearerToken();
      if (!token) throw new Error('Görevleri görmek için siteye giriş yapın.');
      const segment = token.split('.')[1].replace(/-/g, '+').replace(/_/g, '/');
      const userId = JSON.parse(atob(segment.padEnd(Math.ceil(segment.length / 4) * 4, '='))).sub;
      if (!userId) throw new Error('Aktif kullanıcı bulunamadı.');
      if (missionOwner !== userId) { missionRows = []; missionOwner = userId; }
      let setupWarning = '';
      try { await callSupabaseRpc('ensure_user_missions', {}); }
      catch (error) { setupWarning = 'Görev hazırlama başarısız; mevcut görevler gösteriliyor. '; }

      const [missions, progress] = await Promise.all([
        readTable('missions', new URLSearchParams({select:'id,name,description,mission_type,requirement_type,requirement_value,requirement_target,reward_type,reward_amount,mission_order',is_active:'eq.true',order:'mission_order.asc'})),
        readTable('user_missions', new URLSearchParams({select:'id,mission_id,progress,is_completed,is_claimed,reset_at',user_id:'eq.' + userId}))
      ]);
      if (!Array.isArray(missions) || !Array.isArray(progress)) throw new Error('Görev servisi beklenen listeyi döndürmedi.');
      const records = new Map();
      for (const row of progress) {
        const previous = records.get(row.mission_id);
        if (!previous || (Date.parse(row.reset_at) || 0) > (Date.parse(previous.reset_at) || 0)) records.set(row.mission_id, row);
      }
      missionRows = missions.map(mission => {
        const record = records.get(mission.id);
        return record ? {...record, mission, available:true} : {id:'pending-' + mission.id, mission, available:false, progress:null, is_completed:false, is_claimed:false};
      }).sort((a,b)=>a.mission.mission_order-b.mission.mission_order);
      missionsFresh = true;
      ui.missionMessage.textContent = setupWarning + 'Son güncelleme · ' + new Date().toLocaleTimeString('tr-TR');
    } catch (err) {
      missionsFresh = false;
      ui.missionMessage.textContent = err.message + ' · Yenile ile tekrar deneyin. Önceki veriler varsa güncel olmayabilir.';
      log(err.message, 'err');
    } finally {
      missionLoading = false;
      ui.refresh.disabled = false;
      renderMissions();
    }
  }
  async function claimMission(id) {
    const row = missionRows.find(m => m.id === id);
    if (!missionsFresh || !row || row.available === false || !row.is_completed || row.is_claimed || claiming.has(id)) return;
    if (row.reset_at && Date.parse(row.reset_at) <= Date.now()) { await refreshMissions(); return; }
    claiming.add(id);
    renderMissions();
    try {
      const result = await callSupabaseRpc('claim_mission_reward_secure', {p_user_mission_id:id});
      if (!result?.success) throw new Error(result?.error || result?.message || 'Ödül alınamadı.');
      row.is_claimed = true;
      log(row.mission.name + ' · Ödül alındı!', 'ok');
      window.dispatchEvent(new Event('balances-updated'));
      await refreshMissions();
    } catch (err) {
      ui.missionMessage.textContent = err.message;
      log(err.message, 'err');
    } finally {
      claiming.delete(id);
      renderMissions();
    }
  }
  async function missionPulse() {
    if (autoClaimBusy || missionLoading || claiming.size) return;
    autoClaimBusy = true;
    try {
      await refreshMissions();
      if (!state.autoClaim || !missionsFresh) return;
      const ready = missionRows.filter(row => row.available && row.is_completed && !row.is_claimed && (!row.reset_at || Date.parse(row.reset_at) > Date.now())).map(row => row.id);
      for (const id of ready) {
        if (!state.autoClaim || !missionsFresh) break;
        await claimMission(id);
      }
    } finally { autoClaimBusy = false; }
  }
  function renderMissions() {
    if (!ui.missionList) return;

      tab.textContent = (tab.dataset.tab === 'daily' ? 'Günlük' : 'Haftalık') + ' (' + group.filter(row => row.is_completed).length + '/' + group.length + ')';
    });
    const rows = missionRows.filter(row => row.mission.mission_type === missionTab);
    ui.missionList.innerHTML = rows.length ? rows.map(row => {
      const m = row.mission;
      const goal = Math.max(1,Number(m.requirement_value) || 1);
      const progress = Math.max(0,Number(row.estimated ?? row.progress) || 0);
      const pct = Math.min(100,progress / goal * 100);
      const expired = row.reset_at && Date.parse(row.reset_at) <= Date.now();
      const ready = missionsFresh && row.available !== false && row.is_completed && !row.is_claimed && !expired;
      let target = m.requirement_target;
      if (target?.startsWith('rotating_')) {
        const now = new Date();
        const day = Math.floor((Date.UTC(now.getUTCFullYear(),now.getUTCMonth(),now.getUTCDate())-Date.UTC(now.getUTCFullYear(),0,0))/86400000);
        target = ['clicker','memory','math','puzzle','ufo','catch','blockbreaker','match3'][(day+(parseInt(target.split('_')[1]) || 1))%8];
      }
      const gameName = GAMES.find(g => g.id === target)?.name;
      return '<article class="mission ' + (row.is_claimed ? 'claimed' : ready ? 'ready' : '') + '"><div class="card-top"><span class="muted">' + (row.is_claimed?'ALINDI':ready?'ÖDÜL HAZIR':'DEVAM EDİYOR') + '</span><span class="reward">+' + escapeHtml(m.reward_amount) + ' ' + escapeHtml(m.reward_type) + '</span></div><h3>' + escapeHtml(m.name) + '</h3><p>' + escapeHtml(m.description) + (gameName?' · '+escapeHtml(gameName):'') + '</p><div class="progress" role="progressbar" aria-valuemin="0" aria-valuemax="100" aria-valuenow="' + Math.round(pct) + '"><i style="width:' + pct + '%"></i></div><div class="card-top"><span class="muted">' + (row.available === false ? 'İlerleme bekleniyor' : progress + ' / ' + goal + ' · %' + Math.round(pct) + (row.estimated != null ? ' (tahmini)' : '')) + '</span><button class="claim" data-claim="' + escapeHtml(row.id) + '" ' + (!ready || claiming.has(row.id)?'disabled':'') + '>' + (claiming.has(row.id)?'Alınıyor…':row.is_claimed?'✓ Alındı':expired?'Süresi doldu':!missionsFresh?'Yenile':row.available === false?'Bekleniyor':ready?'CLAIM ↗':'Kilitli') + '</button></div></article>';
    }).join('') : '<div class="empty">' + (missionLoading ? 'Görevler yükleniyor…' : 'Görev listesi alınamadı veya bu sekmede görev yok.<br>Aşağıdaki bağlantı durumunu kontrol edip Yenile’ye basın.') + '</div>';
    ui.missionList.querySelectorAll('[data-claim]').forEach(button => button.onclick = () => claimMission(button.dataset.claim));

  function makeDraggable(el, handle, xKey, yKey, x, y) {
    el.style.left = (Number.isFinite(state[xKey]) ? state[xKey] : x) + 'px';
    el.style.top = (Number.isFinite(state[yKey]) ? state[yKey] : y) + 'px';
    clampWindow(el);
    let drag = null;
    el.addEventListener('pointerdown', () => { el.style.zIndex = ++topLayer; });
    handle.addEventListener('pointerdown', event => {
      if (event.button !== 0 || event.target.closest('button')) return;
      const rect = el.getBoundingClientRect();
      drag = {dx:event.clientX-rect.left,dy:event.clientY-rect.top};
      handle.setPointerCapture(event.pointerId);
      event.preventDefault();
    });
    handle.addEventListener('pointermove', event => {
      if (!drag) return;
      el.style.left = (event.clientX-drag.dx) + 'px';
      el.style.top = (event.clientY-drag.dy) + 'px';
      clampWindow(el);
    });
    const finish = () => {
      if (!drag) return;
      drag = null;
      state[xKey] = parseFloat(el.style.left);
      state[yKey] = parseFloat(el.style.top);
      saveState();
    };
    handle.addEventListener('pointerup', finish);
    handle.addEventListener('pointercancel', finish);
    handle.addEventListener('lostpointercapture', finish);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', buildUI);
  } else {
    buildUI();
  }
})();
